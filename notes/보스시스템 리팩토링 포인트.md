`factories` 레지스트리만 유지하고 각 보스 파일이 `window.BossFactories.basic = createBasicBoss` 식으로 등록하면 돼요. 보스 추가할 때 `BossSystem.js`를 건드릴 필요가 없어져요.

#### 2. `spawnIntroT` 래퍼가 `updateBoss`를 덮어씌우는 방식이 위험함

```js
// 현재 — 원본 updateBoss를 클로저로 감싸서 교체
const originalUpdateBoss = boss.updateBoss.bind(boss);
boss.updateBoss = (dt) => {
  if (originalUpdateBoss) originalUpdateBoss(dt);
  // 인트로 로직...
};
```

`bind` + 재할당 패턴은 디버깅 시 콜스택이 불명확해지고, 나중에 `updateBoss`를 또 래핑하면 중첩돼요. 인트로는 보스 객체 프로퍼티로 분리하는 게 맞아요.

```js
// 개선 — 인트로를 별도 처리
function applySpawnIntro(boss) {
  boss.spawnIntroT = BOSS_SPAWN_INTRO_FRAMES;
  boss.spr.alpha = 0;
  boss.spr.scale.set(0.86);
  // updateBoss는 건드리지 않음
}

// 게임 루프의 enemy update 구간에서
function tickBoss(boss, dt) {
  boss.updateBoss(dt);
  if (boss.spawnIntroT > 0) {
    boss.spawnIntroT = Math.max(0, boss.spawnIntroT - dt);
    const eased = 1 - Math.pow(1 - boss.spawnIntroT / BOSS_SPAWN_INTRO_FRAMES, 3);
    boss.spr.alpha = 0.12 + eased * 0.88;
    boss.spr.scale.set(0.86 + eased * 0.14);
  }
}
```

### 🟡 Medium — 중복/일관성 문제

#### 3. `getStealthAwarePlayerTarget` / `getAngleToTrackedPlayer`가 각 보스 클로저 안에서 반복 호출

두 함수는 `BossSystem` 내부에 있는데 각 보스 `updateBoss` 안에서 직접 참조하고 있어요. 문제는 없지만 보스를 파일 분리할 때 **의존성이 암묵적**이라 파악이 어려워요.

```js
// 개선 — createBaseBoss에서 context 객체로 주입
function createBaseBoss(def, options = {}) {
  const ctx = {
    getTarget:       getStealthAwarePlayerTarget,
    getAngleTo:      (x, y) => getAngleToTrackedPlayer(x, y),
    getView:         getCombatView,
  };
  // ...
  boss._ctx = ctx;
  return boss;
}

// 각 보스 updateBoss 안에서
const player = boss._ctx.getTarget();
const view   = boss._ctx.getView();
```

파일 분리 후에도 `BossSystem`의 내부 함수에 직접 접근하지 않아도 돼요.

#### 4. `setState` / `setRedState`가 각 보스마다 중복 정의됨

`KnightBoss`의 `setState`와 `SplitBoss`의 `setRedState`가 **동일한 로직**이에요.

```js
// KnightBoss
function setState(next, duration) {
  boss.state         = next;
  boss.stateTime     = 0;
  boss.stateDuration = duration;
  boss.attackFxPlayed    = false;
  if (next !== "CHARGE") boss.dashTelegraphShown = false;
}

// SplitBoss - Red child
function setRedState(child, next, duration) {
  child.state         = next;
  child.stateTime     = 0;
  child.stateDuration = duration;
  child.attackFxPlayed    = false;
  if (next !== "CHARGE") child.dashTelegraphShown = false;
}
```

`createBaseBoss`에 공용 유틸로 올리면 돼요.

```js
// BossSystem 또는 별도 BossUtils.js
function createStateMachine(entity) {
  return {
    set(next, duration) {
      entity.state              = next;
      entity.stateTime          = 0;
      entity.stateDuration      = duration;
      entity.attackFxPlayed     = false;
      if (next !== "CHARGE") entity.dashTelegraphShown = false;
    },
    tick(dt) { entity.stateTime += dt; },
    elapsed() { return entity.stateTime >= entity.stateDuration; }
  };
}

// 사용
boss.sm = createStateMachine(boss);
boss.sm.set("APPROACH", 34);

// updateBoss 안에서
boss.sm.tick(dt);
if (boss.sm.elapsed()) boss.sm.set("STRAFE", 24);
```

`KnightBoss`와 `SplitBoss` Red child 둘 다 이걸로 대체 가능해요.

#### 8. `SummonerBoss`의 `takeDamage`에서 페이즈 체크 주석이 로직과 불일치

boss.takeDamage = (damage, ctx = {}) => {
