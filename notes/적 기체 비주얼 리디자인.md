# 적 기체 비주얼 리디자인 개발 의뢰서

**프로젝트명** SPIKE ZERO — Enemy Visual Redesign  
**작성일** 2026-04-25  
**대상 파일** `EnemyVisuals.js`

---

## 1. 의뢰 개요

현재 적 기체 드로잉 로직(`drawBasicEnemyShip`)은 단일 함수 내에서 모든 타입을 공통 폴리곤 틀 + 수치 변형 방식으로 처리하고 있어 타입 간 실루엣 차별화가 부족합니다. 이번 의뢰는 **Gunner / Bomber / Tank** 3개 타입에 대해 각각 독립된 드로잉 함수를 적용하고, 전체 비주얼 톤을 **사이버펑크 네온** 방향으로 개선하는 작업입니다.

---

## 2. 디자인 방향

| 항목 | 내용 |
|------|------|
| 스타일 | 사이버펑크 네온 — 얇고 날카로운 실루엣, 어두운 fill, 네온 엣지 발광 |
| 기체 크기 | 기존 `tier.radius` 기준 유지 (변경 없음) |
| 레이어 구조 | `darkFill → circuitLines → edgeGlow → core` 순서 |
| 글로우 처리 | `GlowFilter`는 stroke 전용 레이어에만 적용 (fill 직접 적용 금지) |
| 퀄리티 분기 | `Effects.isHighQuality()` 조건 하에서만 필터 적용 |

### 타입별 컬러 및 실루엣 컨셉

| 타입 | 색상 | 실루엣 |
|------|------|--------|
| Gunner | Gold `#fff0bf` 계열 | 넓적한 동체 + 좌우 독립 포드 + 쌍열 바렐 |
| Bomber | Hot Orange `#ffc187` 계열 | 둔탁한 기수 + 짧은 수평 날개 + 폭탄베이 |
| Tank | Violet `#e3d7ff` 계열 | 정헥사곤 장갑 + 볼트 조인트 + 전방 캐논 |

## 3. 변경 범위

### 3-1. 아키텍처 변경 — `drawBasicEnemyShip` 분기 추가

기존 함수 상단에 타입별 분기를 추가합니다. 나머지 타입의 기존 로직은 **변경하지 않습니다.**

```javascript
function drawBasicEnemyShip(c, tierKey, tier, style) {
  // ↓ 신규 추가 — 3개 타입 전용 함수로 위임
  if (tierKey === "gunner") return drawShip_Gunner(c, tier, style);
  if (tierKey === "bomber") return drawShip_Bomber(c, tier, style);
  if (tierKey === "tank")   return drawShip_Tank(c, tier, style);

  // 기존 로직 유지 (normal / elite / flanker / rusher 등)
  // ...
}
```

### 3-2. 신규 함수 3개 추가

`EnemyVisuals.js` 내 `drawBasicEnemyShip` 함수 **바로 위**에 아래 3개 함수를 삽입합니다.

---

#### `drawShip_Gunner(c, tier, style)`

**실루엣 특징**
- 넓고 평평한 중앙 동체 (8각 폴리곤, 좌우 대칭)
- 상하 독립 포드 — 각각 7각 폴리곤, 내부 accent 라인 포함
- 포드당 쌍열 바렐 2개, 바렐 팁에 발광 닷
- 후방 엔진 타원 글로우

**레이어 구성**

```
aura (ellipse, 7% opacity)
├── pod × 2 (상하 대칭)
│   ├── bridge (커넥터)
│   ├── body (7각 폴리곤, fill #0d0a00)
│   ├── accent (내부 윤곽선, 0.5px)
│   ├── barrels (2열 라인, GlowFilter 적용)
│   ├── tips (바렐 팁 닷)
│   └── podCore (펄스 발광점)
├── body (8각 폴리곤, fill #0a0800)
├── circuit (수평/수직 내부선, 0.55px)
├── cockpit (5각 윤곽, fill 7%)
├── edgeTop / edgeBot (엣지 하이라이트, GlowFilter)
├── eng (후방 타원 글로우)
└── core (3단 원 — 아우라/컬러/화이트)
```

**구현 코드**

```javascript
function drawShip_Gunner(c, tier, style) {
  const r = tier.radius;
  const col = tier.fillColor;
  const line = tier.lineColor;

  function makePod(sign) {
    const g = new PIXI.Container();

    const body = new PIXI.Graphics();
    body.beginFill(0x0d0a00, 0.97);
    body.lineStyle(1.2, line, 0.9);
    body.drawPolygon([
       r*0.42, sign*(-r*0.52),
       r*0.12, sign*(-r*0.52),
       r*0.05, sign*(-r*0.62),
       r*0.12, sign*(-r*0.84),
       r*0.42, sign*(-r*0.84),
       r*0.48, sign*(-r*0.74),
       r*0.48, sign*(-r*0.62),
    ]);
    body.endFill();

    const accent = new PIXI.Graphics();
    accent.lineStyle(0.5, line, 0.4);
    accent.drawPolygon([
       r*0.38, sign*(-r*0.56),
       r*0.16, sign*(-r*0.56),
       r*0.12, sign*(-r*0.64),
       r*0.16, sign*(-r*0.80),
       r*0.38, sign*(-r*0.80),
       r*0.42, sign*(-r*0.72),
       r*0.42, sign*(-r*0.64),
    ]);

    const barrels = new PIXI.Graphics();
    barrels.lineStyle(2.2, line, 0.95, 0.5, true);
    barrels.moveTo(r*0.48, sign*(-r*0.65));
    barrels.lineTo(r*0.82, sign*(-r*0.65));
    barrels.moveTo(r*0.48, sign*(-r*0.74));
    barrels.lineTo(r*0.78, sign*(-r*0.74));

    const tips = new PIXI.Graphics();
    tips.beginFill(line, 1);
    tips.drawCircle(r*0.82, sign*(-r*0.65), r*0.026);
    tips.drawCircle(r*0.78, sign*(-r*0.74), r*0.026);
    tips.endFill();

    const core = new PIXI.Graphics();
    core.beginFill(col, 0.8);
    core.drawCircle(r*0.26, sign*(-r*0.68), r*0.055);
    core.endFill();
    core.beginFill(0xffffff, 0.6);
    core.drawCircle(r*0.26, sign*(-r*0.68), r*0.022);
    core.endFill();

    const bridge = new PIXI.Graphics();
    bridge.beginFill(0x0d0a00, 0.7);
    bridge.lineStyle(0.8, line, 0.3);
    bridge.drawRoundedRect(
      -r*0.06,
      sign*(sign < 0 ? -r*0.52 : r*0.42),
      r*0.52, r*0.1, 2
    );
    bridge.endFill();

    if (Effects.isHighQuality()) {
      const glowLine = Effects.makeGlowFilter({ color: line, distance: 10, outerStrength: 1.4 });
      if (glowLine) barrels.filters = [glowLine];
      const glowCore = Effects.makeGlowFilter({ color: col, distance: 8, outerStrength: 2 });
      if (glowCore) core.filters = [glowCore];
    }

    g.addChild(bridge, body, accent, barrels, tips, core);
    return g;
  }

  const aura = new PIXI.Graphics();
  aura.beginFill(col, 0.07);
  aura.drawEllipse(0, 0, r*1.35, r*0.65);
  aura.endFill();

  const body = new PIXI.Graphics();
  body.beginFill(0x0a0800, 0.97);
  body.lineStyle(1.6, line, 1);
  body.drawPolygon([
     r*0.88,  0,
     r*0.66, -r*0.25,
     r*0.20, -r*0.38,
    -r*0.74, -r*0.30,
    -r*0.88,  0,
    -r*0.74,  r*0.30,
     r*0.20,  r*0.38,
     r*0.66,  r*0.25,
  ]);
  body.endFill();

  const circuit = new PIXI.Graphics();
  circuit.lineStyle(0.55, line, 0.28);
  circuit.moveTo(-r*0.66, 0); circuit.lineTo(r*0.64, 0);
  circuit.moveTo(r*0.28, -r*0.3); circuit.lineTo(r*0.28, r*0.3);
  circuit.moveTo(-r*0.22, -r*0.34); circuit.lineTo(-r*0.22, r*0.34);

  const cockpit = new PIXI.Graphics();
  cockpit.lineStyle(0.8, line, 0.55);
  cockpit.beginFill(col, 0.07);
  cockpit.drawPolygon([
    r*0.74, 0, r*0.52, -r*0.16, r*0.14, -r*0.20,
    r*0.14, r*0.20, r*0.52, r*0.16,
  ]);
  cockpit.endFill();

  const edgeTop = new PIXI.Graphics();
  edgeTop.lineStyle(1.2, line, 0.75);
  edgeTop.moveTo( r*0.88, 0);
  edgeTop.lineTo( r*0.66, -r*0.25);
  edgeTop.lineTo( r*0.20, -r*0.38);
  edgeTop.lineTo(-r*0.74, -r*0.30);

  const edgeBot = new PIXI.Graphics();
  edgeBot.lineStyle(1.2, line, 0.75);
  edgeBot.moveTo( r*0.88, 0);
  edgeBot.lineTo( r*0.66,  r*0.25);
  edgeBot.lineTo( r*0.20,  r*0.38);
  edgeBot.lineTo(-r*0.74,  r*0.30);

  const eng = new PIXI.Graphics();
  eng.beginFill(col, 0.35);
  eng.drawEllipse(-r*0.88, 0, r*0.06, r*0.18);
  eng.endFill();

  const core = new PIXI.Graphics();
  core.beginFill(col, 0.2);
  core.drawCircle(0, 0, r*0.16);
  core.endFill();
  core.beginFill(col, 0.82);
  core.drawCircle(0, 0, r*0.09);
  core.endFill();
  core.beginFill(0xffffff, 0.6);
  core.drawCircle(0, 0, r*0.036);
  core.endFill();

  if (Effects.isHighQuality()) {
    const gEdge = Effects.makeGlowFilter({ color: line, distance: 12, outerStrength: 1.2 });
    if (gEdge) { edgeTop.filters = [gEdge]; edgeBot.filters = [gEdge]; }
    const gCore = Effects.makeGlowFilter({ color: col, distance: 14, outerStrength: 2.2 });
    if (gCore) core.filters = [gCore];
    const gEng  = Effects.makeGlowFilter({ color: col, distance: 10, outerStrength: 1.6 });
    if (gEng)  eng.filters  = [gEng];
  }

  c.addChild(aura, makePod(-1), makePod(1), body, circuit, cockpit, edgeTop, edgeBot, eng, core);
}
```

#### `drawShip_Bomber(c, tier, style)`

**실루엣 특징**
- 둔탁한 기수 + 길고 넓은 동체 (9각 폴리곤)
- 짧고 날카로운 삼각 날개 (기수 근처, 윙팁 발광)
- 직각 폭탄베이 패널 + 4개 페이로드 블록
- 후방 이중 노즐 (수직 2개)

**레이어 구성**

```
aura (ellipse)
├── wings (상하 삼각, fill #0e0600)
├── wingEdge (날개 엣지 라인, GlowFilter)
├── wingtip (닷 × 2)
├── body (9각 폴리곤, fill #0e0600)
├── panels (동체 내부 패널선)
├── noseEdge / noseEdgeB (기수 엣지, GlowFilter)
├── bay (폭탄베이 직각 패널)
├── eng (노즐 박스 × 2)
├── nozzleGlow (노즐 발광, GlowFilter)
└── core
```

**구현 코드**

```javascript
function drawShip_Bomber(c, tier, style) {
  const r = tier.radius;
  const col = tier.fillColor;
  const line = tier.lineColor;

  const aura = new PIXI.Graphics();
  aura.beginFill(col, 0.07);
  aura.drawEllipse(0, 0, r*0.9, r*1.18);
  aura.endFill();

  const wings = new PIXI.Graphics();
  wings.beginFill(0x0e0600, 0.96);
  wings.lineStyle(1.1, line, 0.85);
  wings.drawPolygon([ r*0.22,-r*0.32, r*1.12,-r*0.6, r*1.04,-r*0.44, r*0.22,-r*0.22 ]);
  wings.drawPolygon([ r*0.22, r*0.32, r*1.12, r*0.6,  r*1.04, r*0.44, r*0.22, r*0.22 ]);
  wings.endFill();

  const wingEdge = new PIXI.Graphics();
  wingEdge.lineStyle(1.3, line, 0.85);
  wingEdge.moveTo(r*0.22, -r*0.32); wingEdge.lineTo(r*1.12, -r*0.6);
  wingEdge.moveTo(r*0.22,  r*0.32); wingEdge.lineTo(r*1.12,  r*0.6);

  const wingtip = new PIXI.Graphics();
  wingtip.beginFill(line, 1);
  wingtip.drawCircle(r*1.08, -r*0.54, r*0.03);
  wingtip.drawCircle(r*1.08,  r*0.54, r*0.03);
  wingtip.endFill();

  const body = new PIXI.Graphics();
  body.beginFill(0x0e0600, 0.98);
  body.lineStyle(1.8, line, 0.92);
  body.drawPolygon([
     r*0.68,  0,
     r*0.46, -r*0.20,
     r*0.10, -r*0.40,
    -r*0.74, -r*0.48,
    -r*0.86, -r*0.20,
    -r*0.86,  r*0.20,
    -r*0.74,  r*0.48,
     r*0.10,  r*0.40,
     r*0.46,  r*0.20,
  ]);
  body.endFill();

  const panels = new PIXI.Graphics();
  panels.lineStyle(0.5, line, 0.24);
  panels.moveTo(-r*0.44, -r*0.46); panels.lineTo(r*0.32, -r*0.22);
  panels.moveTo(-r*0.44,  r*0.46); panels.lineTo(r*0.32,  r*0.22);
  panels.moveTo(-r*0.16, -r*0.44); panels.lineTo(-r*0.16, r*0.44);

  const noseEdge = new PIXI.Graphics();
  noseEdge.lineStyle(1.4, line, 0.82);
  noseEdge.moveTo(r*0.68, 0);
  noseEdge.lineTo(r*0.46, -r*0.20);
  noseEdge.lineTo(r*0.10, -r*0.40);

  const noseEdgeB = new PIXI.Graphics();
  noseEdgeB.lineStyle(1.4, line, 0.82);
  noseEdgeB.moveTo(r*0.68, 0);
  noseEdgeB.lineTo(r*0.46,  r*0.20);
  noseEdgeB.lineTo(r*0.10,  r*0.40);

  const bay = new PIXI.Graphics();
  bay.beginFill(col, 0.07);
  bay.lineStyle(1, line, 0.6);
  bay.drawRect(r*0.04, -r*0.26, r*0.52, r*0.52);
  bay.endFill();
  bay.lineStyle(0.7, line, 0.45);
  bay.moveTo(r*0.30, -r*0.26); bay.lineTo(r*0.30, r*0.26);
  bay.lineStyle(0);
  bay.beginFill(col, 0.4);
  bay.drawRect(r*0.09, -r*0.18, r*0.14, r*0.10);
  bay.drawRect(r*0.09,  r*0.08, r*0.14, r*0.10);
  bay.drawRect(r*0.35, -r*0.18, r*0.14, r*0.10);
  bay.drawRect(r*0.35,  r*0.08, r*0.14, r*0.10);
  bay.endFill();

  const eng = new PIXI.Graphics();
  eng.beginFill(0x0e0600, 0.95);
  eng.lineStyle(0.9, line, 0.8);
  eng.drawRect(-r*0.86, -r*0.32, r*0.16, r*0.24);
  eng.drawRect(-r*0.86,  r*0.08, r*0.16, r*0.24);
  eng.endFill();

  const nozzleGlow = new PIXI.Graphics();
  nozzleGlow.beginFill(col, 0.6);
  nozzleGlow.drawRect(-r*0.87, -r*0.30, r*0.045, r*0.20);
  nozzleGlow.drawRect(-r*0.87,  r*0.10, r*0.045, r*0.20);
  nozzleGlow.endFill();

  const core = new PIXI.Graphics();
  core.beginFill(col, 0.18);
  core.drawCircle(-r*0.14, 0, r*0.18);
  core.endFill();
  core.beginFill(col, 0.82);
  core.drawCircle(-r*0.14, 0, r*0.10);
  core.endFill();
  core.beginFill(0xffffff, 0.6);
  core.drawCircle(-r*0.14, 0, r*0.04);
  core.endFill();

  if (Effects.isHighQuality()) {
    const gw = Effects.makeGlowFilter({ color: line, distance: 10, outerStrength: 1.3 });
    if (gw) { wingEdge.filters = [gw]; noseEdge.filters = [gw]; noseEdgeB.filters = [gw]; }
    const gn = Effects.makeGlowFilter({ color: col, distance: 8, outerStrength: 1.8 });
    if (gn) { nozzleGlow.filters = [gn]; wingtip.filters = [gn]; }
    const gc = Effects.makeGlowFilter({ color: col, distance: 14, outerStrength: 2.4 });
    if (gc) core.filters = [gc];
  }

  c.addChild(aura, wings, body, panels, wingEdge, wingtip, noseEdge, noseEdgeB, bay, eng, nozzleGlow, core);
}
```

#### `drawShip_Tank(c, tier, style)`

**실루엣 특징**
- 정헥사곤 메인 바디 (flat-top 기준, `Math.PI/3` 간격 6꼭짓점)
- 외곽 헥사곤 링 (dim, 0.22 opacity)
- 6개 장갑 면 개별 fill + 볼트 닷 + 스포크 라인
- 이너 헥사곤 (2차 장갑)
- 우측 전방 다이아몬드형 캐논 + 배럴 라인 + 머즐 닷
- 좌측 후방 벤트 4개 + 캡 닷
- 상단 회로 트레이스 (분기형)

**레이어 구성**

```
aura (hex shape)
├── outerRing (외곽 링, 0.6px dim)
├── faces (6개 면 fill, 교번 명도)
├── body (메인 헥사곤, fill #0a0014, GlowFilter)
├── inner (이너 헥사곤)
├── bolts (꼭짓점 닷 + 스포크)
├── trace (회로 트레이스)
├── cannon (다이아몬드 + 배럴, GlowFilter)
├── vents (수평 라인 × 4, GlowFilter)
├── ventCaps (벤트 캡 닷)
└── core
```

**구현 코드**

```javascript
function drawShip_Tank(c, tier, style) {
  const r = tier.radius;
  const col = tier.fillColor;
  const line = tier.lineColor;

  function hex(rad, rot=0) {
    const pts = [];
    for (let i=0; i<6; i++) {
      const a = Math.PI/3 * i + rot;
      pts.push(rad*Math.cos(a), rad*Math.sin(a));
    }
    return pts;
  }

  const aura = new PIXI.Graphics();
  aura.beginFill(col, 0.07);
  aura.drawPolygon(hex(r*1.32));
  aura.endFill();

  const outerRing = new PIXI.Graphics();
  outerRing.lineStyle(0.6, line, 0.22);
  outerRing.drawPolygon(hex(r*1.14));

  const H = hex(r);
  const faces = new PIXI.Graphics();
  faces.lineStyle(0.9, line, 0.85);
  for (let i=0; i<6; i++) {
    const ax = H[i*2], ay = H[i*2+1];
    const bx = H[((i+1)%6)*2], by = H[((i+1)%6)*2+1];
    faces.beginFill(col, i % 2 === 0 ? 0.04 : 0.02);
    faces.drawPolygon([0,0, ax,ay, (ax+bx)/2*1.01,(ay+by)/2*1.01, bx,by]);
    faces.endFill();
  }

  const body = new PIXI.Graphics();
  body.beginFill(0x0a0014, 0.98);
  body.lineStyle(1.8, line, 0.94);
  body.drawPolygon(hex(r));
  body.endFill();

  const inner = new PIXI.Graphics();
  inner.beginFill(col, 0.06);
  inner.lineStyle(1.0, line, 0.5);
  inner.drawPolygon(hex(r*0.62));
  inner.endFill();

  const bolts = new PIXI.Graphics();
  for (let i=0; i<6; i++) {
    const a = Math.PI/3 * i;
    const bx = r*Math.cos(a), by = r*Math.sin(a);
    const ix = r*0.62*Math.cos(a), iy = r*0.62*Math.sin(a);
    bolts.lineStyle(0.7, line, 0.38);
    bolts.moveTo(bx, by); bolts.lineTo(ix, iy);
    bolts.lineStyle(0);
    bolts.beginFill(line, 0.62);
    bolts.drawCircle(bx, by, r*0.055);
    bolts.endFill();
  }

  const cannon = new PIXI.Graphics();
  cannon.beginFill(0x0a0014, 0.96);
  cannon.lineStyle(1.2, line, 0.92);
  cannon.drawPolygon([ r,0, r*1.12,-r*0.09, r*1.38,0, r*1.12,r*0.09 ]);
  cannon.endFill();
  cannon.lineStyle(1.8, line, 0.95);
  cannon.moveTo(r, 0); cannon.lineTo(r*1.38, 0);
  cannon.lineStyle(0);
  cannon.beginFill(line, 1);
  cannon.drawCircle(r*1.38, 0, r*0.042);
  cannon.endFill();
  cannon.beginFill(0xffffff, 0.7);
  cannon.drawCircle(r*1.38, 0, r*0.018);
  cannon.endFill();

  const vents = new PIXI.Graphics();
  vents.lineStyle(1.8, line, 0.65, 0.5, true);
  const ventOffsets = [-r*0.22, -r*0.09, r*0.04, r*0.17];
  ventOffsets.forEach(vy => {
    vents.moveTo(-r, vy);
    vents.lineTo(-r*1.2, vy);
  });

  const ventCaps = new PIXI.Graphics();
  ventCaps.beginFill(line, 0.7);
  ventOffsets.forEach(vy => {
    ventCaps.drawCircle(-r*1.2, vy, r*0.022);
  });
  ventCaps.endFill();

  const trace = new PIXI.Graphics();
  trace.lineStyle(0.6, line, 0.4);
  trace.moveTo(0, -r*0.62); trace.lineTo(0, -r*0.3);
  trace.lineTo(r*0.22, -r*0.22); trace.lineTo(r*0.22, 0);
  trace.moveTo(0, -r*0.3);
  trace.lineTo(-r*0.22, -r*0.22); trace.lineTo(-r*0.22, 0);

  const core = new PIXI.Graphics();
  core.beginFill(col, 0.18);
  core.drawCircle(0, 0, r*0.26);
  core.endFill();
  core.beginFill(col, 0.75);
  core.drawCircle(0, 0, r*0.13);
  core.endFill();
  core.beginFill(0xffffff, 0.65);
  core.drawCircle(0, 0, r*0.05);
  core.endFill();

  if (Effects.isHighQuality()) {
    const gBody = Effects.makeGlowFilter({ color: line, distance: 11, outerStrength: 1.2 });
    if (gBody) body.filters = [gBody];
    const gCan  = Effects.makeGlowFilter({ color: line, distance: 10, outerStrength: 1.6 });
    if (gCan)  cannon.filters = [gCan];
    const gVent = Effects.makeGlowFilter({ color: col,  distance: 8,  outerStrength: 1.4 });
    if (gVent) { vents.filters = [gVent]; ventCaps.filters = [gVent]; }
    const gCore = Effects.makeGlowFilter({ color: col,  distance: 16, outerStrength: 2.4 });
    if (gCore) core.filters = [gCore];
  }

  c.addChild(aura, outerRing, faces, body, inner, bolts, trace, cannon, vents, ventCaps, core);
}
```
